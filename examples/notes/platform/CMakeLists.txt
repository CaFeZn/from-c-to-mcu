cmake_minimum_required(VERSION 3.16)
project(platform_examples C)
add_executable(platform_test test.c)
set_target_properties(platform_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(platform_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
endif()
enable_testing()
add_test(NAME platform_model COMMAND platform_test)
