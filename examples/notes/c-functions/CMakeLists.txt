cmake_minimum_required(VERSION 3.16)
project(c_functions_notes LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()

function(note_program name)
    add_executable(${name} ${ARGN})
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        # These teaching executables use assert as their test oracle.
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endfunction()

note_program(pointer_demo pointer_demo.c)
note_program(stack_demo stack_demo.c)
note_program(objects_demo objects_demo.c)
note_program(tables_demo tables_demo.c)
note_program(patterns_demo patterns_demo.c)
note_program(abi_demo abi_demo.c abi_targets.c)
note_program(embedded_cpp_demo embedded_cpp_demo.cpp cpp_c_api.c)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    note_program(embedded_cpp_noeh embedded_cpp_demo.cpp cpp_c_api.c)
    target_compile_options(embedded_cpp_noeh PRIVATE
        "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions;-fno-rtti>")
endif()
