cmake_minimum_required(VERSION 3.16)
project(core_note_models LANGUAGES C)
add_executable(core_note_models test.c)
set_target_properties(core_note_models PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(core_note_models PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME core_note_models COMMAND core_note_models)

foreach(snippet IN ITEMS read-modify-write vector-table interrupt-storms exclusive-access)
  add_executable(${snippet} snippets/${snippet}.c)
  set_target_properties(${snippet} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${snippet} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  endif()
  add_test(NAME ${snippet} COMMAND ${snippet})
endforeach()
