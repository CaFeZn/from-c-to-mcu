cmake_minimum_required(VERSION 3.16)
project(analog_note_models LANGUAGES C)
add_executable(analog_note_tests test.c)
set_target_properties(analog_note_tests PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(MSVC)
  target_compile_options(analog_note_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(analog_note_tests PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
  target_link_libraries(analog_note_tests PRIVATE m)
endif()
enable_testing()
add_test(NAME analog_note_models COMMAND analog_note_tests)
