cmake_minimum_required(VERSION 3.20)
project(workflow_examples C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

add_executable(hello src/hello.c)
add_executable(greet src/greet.c)
add_executable(hello_c_lab snapshots/hello-c-lab.c)
add_executable(hello_again snapshots/hello-again.c)

enable_testing()
add_test(NAME hello COMMAND hello)
add_test(NAME greet COMMAND greet)
add_test(NAME hello_c_lab COMMAND hello_c_lab)
add_test(NAME hello_again COMMAND hello_again)
set_tests_properties(hello PROPERTIES PASS_REGULAR_EXPRESSION "hello, world")
set_tests_properties(greet PROPERTIES PASS_REGULAR_EXPRESSION "hi, c-lab")
set_tests_properties(hello_c_lab PROPERTIES PASS_REGULAR_EXPRESSION "hello, c-lab")
set_tests_properties(hello_again PROPERTIES PASS_REGULAR_EXPRESSION "hello, again")
