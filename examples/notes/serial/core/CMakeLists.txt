cmake_minimum_required(VERSION 3.16)
project(serial_core C)
add_executable(serial_core test.c)
set_target_properties(serial_core PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(serial_core PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
elseif(MSVC)
  target_compile_options(serial_core PRIVATE /UNDEBUG)
endif()
enable_testing()
add_test(NAME serial_core COMMAND serial_core)
