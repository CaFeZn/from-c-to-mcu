cmake_minimum_required(VERSION 3.16)
project(serial_hal_flow_models C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
foreach(model IN ITEMS callback flow)
    add_executable(${model}_model ${model}_model.c)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${model}_model PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_test(NAME ${model}_model COMMAND ${model}_model)
endforeach()
