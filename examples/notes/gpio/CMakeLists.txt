cmake_minimum_required(VERSION 3.16)
project(gpio_note_examples C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(gpio_notes_test test.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(gpio_notes_test PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
  target_link_libraries(gpio_notes_test PRIVATE m)
elseif(MSVC)
  target_compile_options(gpio_notes_test PRIVATE /W4 /UNDEBUG)
endif()
enable_testing()
add_test(NAME gpio_note_models COMMAND gpio_notes_test)
