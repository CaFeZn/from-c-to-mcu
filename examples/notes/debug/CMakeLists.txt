cmake_minimum_required(VERSION 3.16)
project(debug_note_models C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(debug_models test.c)
add_executable(debug_optimization optimization.c)
enable_testing()
add_test(NAME debug_models COMMAND debug_models)
add_test(NAME debug_optimization COMMAND debug_optimization)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(debug_models PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
  target_compile_options(debug_optimization PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
