cmake_minimum_required(VERSION 3.16)
project(timers_notes C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_executable(timers-core core/timers-core-test.c)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(timers-core PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
endif()
add_test(NAME timers-core COMMAND timers-core)
foreach(name IN ITEMS scheduler capture jitter test)
  add_executable(timers-${name} ${name}.c)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(timers-${name} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
  endif()
  if(NOT MSVC)
    target_link_libraries(timers-${name} PRIVATE m)
  endif()
  add_test(NAME timers-${name} COMMAND timers-${name})
endforeach()
