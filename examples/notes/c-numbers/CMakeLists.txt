cmake_minimum_required(VERSION 3.16)
project(course_c_numbers LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
foreach(name integer-conversions integer-overflow float-fixed-point complement-saturation randomness)
    add_executable(${name} ${name}.c)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wconversion -Werror -pedantic -UNDEBUG)
    elseif(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endforeach()
